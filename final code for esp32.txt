
#include <WiFi.h>

  }
  Serial.println();

  Serial.println(WiFi.localIP());

}


void loop() {
    if (WiFi.status() == WL_CONNECTED) {
        int flexValue = analogRead(26); // Read your flex sensor here

        // Construct the request URL with the sensor value as a parameter
        String url = String(serverAddress) + "?sensor_value=" + String(flexValue);

        HTTPClient http;
        http.begin(url); // Specify the complete URL with parameters

        int httpResponseCode = http.GET(); // Send the GET request

        if (httpResponseCode > 0) {
            String response = http.getString();
            Serial.println("HTTP Response code: " + String(httpResponseCode));
            Serial.println("Response: " + response);
        } else {
            Serial.println("Error in HTTP GET request");
        }

        http.end();
    }
    delay(5000); // Adjust delay as needed before sending the next data
}
